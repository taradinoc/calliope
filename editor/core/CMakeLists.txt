cmake_minimum_required(VERSION 3.20)
project(calliope_editor_core C)
set(CMAKE_C_STANDARD 11)
if(NOT CORE_SOURCE)
  message(FATAL_ERROR "Set CORE_SOURCE to the pinned Genesis Plus GX checkout")
endif()
set(dirs core core/z80 core/m68k core/ntsc core/sound core/sound/minimp3 core/input_hw core/cd_hw core/cart_hw core/cart_hw/svp core/sound/tremor)
foreach(dir IN LISTS dirs)
  file(GLOB files "${CORE_SOURCE}/${dir}/*.c")
  list(APPEND sources ${files})
  list(APPEND includes "${CORE_SOURCE}/${dir}")
endforeach()
set(common ${CORE_SOURCE}/libretro/libretro-common)
set(common_files streams/file_stream streams/file_stream_transforms compat/fopen_utf8 compat/compat_snprintf compat/compat_strl compat/compat_strcasestr compat/compat_posix_string encodings/encoding_utf file/file_path file/retro_dirent lists/string_list lists/dir_list memmap/memalign string/stdstring vfs/vfs_implementation)
foreach(file IN LISTS common_files)
  list(APPEND sources "${common}/${file}.c")
endforeach()
foreach(file adler32 crc32 inffast inflate inftrees zutil)
  list(APPEND sources "${CORE_SOURCE}/libretro/deps/zlib-1.2.11/${file}.c")
endforeach()
# Instrument a build-directory copy; preserve the clean pinned upstream tree.
set(cpu_source "${CORE_SOURCE}/core/m68k/m68kcpu.c")
file(READ "${cpu_source}" cpu)
set(loop "  while (m68k.cycles < cycles)\n  {")
string(FIND "${cpu}" "${loop}" loop_at)
if(loop_at EQUAL -1)
  message(FATAL_ERROR "Pinned 68000 profiler insertion point missing")
endif()
string(REPLACE "${loop}" "${loop}\n    unsigned int clp_start = m68k.cycles;\n    int clp_busy = clp_editor_cpu_busy();" cpu "${cpu}")
set(trace "    m68ki_exception_if_trace(); /* auto-disable (see m68kcpu.h) */")
string(FIND "${cpu}" "${trace}" trace_at)
if(trace_at EQUAL -1)
  message(FATAL_ERROR "Pinned 68000 profiler end point missing")
endif()
string(REPLACE "${trace}" "${trace}\n    if (clp_busy) clp_editor_cpu_add(m68k.cycles - clp_start);" cpu "${cpu}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/m68kcpu-profile.c" "extern int clp_editor_cpu_busy(void);\nextern void clp_editor_cpu_add(unsigned int);\n${cpu}")
list(REMOVE_ITEM sources "${cpu_source}")
list(APPEND sources "${CMAKE_CURRENT_BINARY_DIR}/m68kcpu-profile.c")
add_library(genesis_plus_gx SHARED ${sources} "${CORE_SOURCE}/libretro/libretro.c" telemetry.c)
set_target_properties(genesis_plus_gx PROPERTIES PREFIX "")
target_include_directories(genesis_plus_gx PRIVATE ${includes} "${CORE_SOURCE}/libretro" "${common}/include")
target_compile_definitions(genesis_plus_gx PRIVATE USE_LIBTREMOR USE_PER_SOUND_CHANNELS_CONFIG USE_16BPP_RENDERING FRONTEND_SUPPORTS_RGB565 LSB_FIRST BYTE_ORDER=LITTLE_ENDIAN __LIBRETRO__ USE_LIBRETRO_VFS M68K_OVERCLOCK_SHIFT=20 Z80_OVERCLOCK_SHIFT=20 HAVE_YM3438_CORE HAVE_OPLL_CORE MAXROMSIZE=10485760)
if(MSVC)
  target_compile_options(genesis_plus_gx PRIVATE /utf-8)
  target_compile_definitions(genesis_plus_gx PRIVATE "INLINE=static _inline" _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(genesis_plus_gx PRIVATE winmm ws2_32 shlwapi)
else()
  target_compile_definitions(genesis_plus_gx PRIVATE "INLINE=static inline" HAVE_ZLIB HAVE_UNISTD_H)
  target_link_libraries(genesis_plus_gx PRIVATE m)
endif()
